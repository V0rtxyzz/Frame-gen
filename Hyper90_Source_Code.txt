        "${ROOT_DIR}/src/FrameBlend.cpp" \
        "${ROOT_DIR}/src/Modules.cpp" \
        -o "/tmp/test_hyper90"
    /tmp/test_hyper90
    /tmp/test_hyper90 --generate-config "${ROOT_DIR}/config"
fi

echo "==> [2/4] Compiling Android arm64-v8a shared library (libhyper90_fps_booster.so)..."

    cmake -S "${ROOT_DIR}" -B "${BUILD_DIR}" -GNinja \
        -DCMAKE_BUILD_TYPE=Release \
        -DCMAKE_TOOLCHAIN_FILE="${NDK_DIR}/build/cmake/android.toolchain.cmake" \
        -DANDROID_ABI=arm64-v8a \
        -DANDROID_PLATFORM=android-26 \

    cmake --build "${BUILD_DIR}" --target hyper90_fps_booster -j"$(nproc)"
    cp "${BUILD_DIR}/libhyper90_fps_booster.so" "${RELEASE_DIR}/libhyper90_fps_booster.so"
elif [[ -f "/tmp/tools/android-libc.txt" ]]; then
    export ZIG_LIBC="/tmp/tools/android-libc.txt"
    python3 -m ziglang c++ -target aarch64-linux-android \
        -D__ANDROID_API__=30 \
        -std=c++20 -Os -flto -shared -fPIC \
        -fvisibility=hidden -fvisibility-inlines-hidden \
        -ffunction-sections -fdata-sections \
        -isystem /tmp/tools/pwn-ndk-sysroot/include \
        -isystem /tmp/tools/pwn-ndk-sysroot/include/aarch64-linux-android \
        -I"${ROOT_DIR}/include" \
        -I"/tmp/ref/preloader-android/include" \
        -I"/tmp/tools/fmt/include" \
        -I"/tmp/tools/json/include" \
        -DFMT_HEADER_ONLY \
        "${ROOT_DIR}/src/Config.cpp" \
        "${ROOT_DIR}/src/Telemetry.cpp" \
        "${ROOT_DIR}/src/EngineOptionsTuner.cpp" \
        "${ROOT_DIR}/src/SystemOptimizers.cpp" \
        "${ROOT_DIR}/src/NativeHooks.cpp" \
        "${ROOT_DIR}/src/FrameBlend.cpp" \
        "${ROOT_DIR}/src/Modules.cpp" \
        "${ROOT_DIR}/src/Hyper90Mod.cpp" \
        -L/tmp/tools/pwn-ndk-sysroot/lib \
        -Wl,--no-as-needed \
        -lpreloader -llog -landroid -lEGL -lGLESv2 -lGLESv3 -lm -ldl -lbionic_c \
        -Wl,--version-script=/tmp/tools/exports.map \
        -Wl,--gc-sections \
        -Wl,-s \
        -Wl,--allow-shlib-undefined \
        -Wl,-z,max-page-size=16384 \
        -Wl,-soname,libhyper90_fps_booster.so \
        -o "${RELEASE_DIR}/libhyper90_fps_booster.so"
else
    echo "Error: Neither ANDROID_NDK_HOME nor /tmp/tools/android-libc.txt found." >&2
    exit 1
fi

echo "==> [3/4] Packaging Hyper90FPSBooster.levipack..."
STAGE_DIR="/tmp/hyper90-levipack-stage"
rm -rf "${STAGE_DIR}"
mkdir -p "${STAGE_DIR}/config"

cp "${ROOT_DIR}/manifest.json" "${STAGE_DIR}/manifest.json"
cp "${RELEASE_DIR}/libhyper90_fps_booster.so" "${STAGE_DIR}/libhyper90_fps_booster.so"
cp "${ROOT_DIR}/config/config.json" "${STAGE_DIR}/config/config.json"
cp "${ROOT_DIR}/config/config.schema.json" "${STAGE_DIR}/config/config.schema.json"

python3 - <<'PY'
import os, zipfile

stage = "/tmp/hyper90-levipack-stage"
outputs = [
    os.path.abspath("Hyper90FPSBooster.levipack"),
    os.path.abspath("release/Hyper90FPSBooster.levipack"),
]

for out_path in outputs:
    if os.path.exists(out_path):
        os.remove(out_path)
    with zipfile.ZipFile(out_path, "w", compression=zipfile.ZIP_DEFLATED, compresslevel=9) as zf:
        for root, _, files in os.walk(stage):
            for f in sorted(files):
                full_path = os.path.join(root, f)
                rel_path = os.path.relpath(full_path, stage).replace(os.sep, "/")
                zf.write(full_path, rel_path)
    print(f"Created {out_path} ({os.path.getsize(out_path)} bytes)")
PY

echo "==> [4/4] Verifying .levipack package & ELF64 AArch64 binary..."
readelf -h "${RELEASE_DIR}/libhyper90_fps_booster.so" | grep -E "Class:|Machine:|Type:"
nm -D --defined-only "${RELEASE_DIR}/libhyper90_fps_booster.so"
echo "Build complete!"

#pragma once

#include <array>
#include <atomic>
#include <cstdint>

namespace hyper90 {

enum class GovernorTier : int {
    Locked90Plus = 0, // >= 89.5 FPS: Rock-solid 90 FPS locked
    Balanced90   = 1, // 84.0 - 89.5 FPS: Mild dynamic LOD & mipmap boost
    Aggressive90 = 2, // 75.0 - 84.0 FPS: Particle budget & micro-batch culling active
    Ultra90Lock  = 3  // < 75.0 FPS or manual Ultra override: Maximum 90-FPS rescue
};

struct FrameSnapshot {
    float        avgFps{90.0f};
    float        onePercentLowFps{90.0f};
    float        lastFrameMs{11.11f};
    float        jitterMs{0.0f};
    GovernorTier tier{GovernorTier::Locked90Plus};
    uint64_t     totalFrames{0};
    uint64_t     glStateDedupSaved{0};
    uint64_t     tbdrTilesDiscarded{0};
    uint64_t     drawCallsCulled{0};
    uint64_t     glFinishSuppressed{0};
    uint64_t     textureFiltersOptimized{0};
    uint64_t     generatedFrames{0};
    uint64_t     hudCacheHits{0};
    int          lastFrameDrawCalls{0};
    int          lastFrameCulledDraws{0};
    int          pinnedBigCores{0};
    int          optionsFilesTuned{0};
};

class alignas(64) TelemetryEngine {
public:
    static constexpr size_t kWindowSize = 120;

    TelemetryEngine() noexcept;

    // Called inside eglSwapBuffers: paces the frame to targetFps and records metrics.
    void paceAndRecordFrame() noexcept;

    // Direct frame recording for deterministic testing with synthetic timestamps.
    void recordFrameDurationNs(int64_t durationNs) noexcept;

    // Module 13: records one extra presented (generated) frame in the FPS window without
    // touching per-frame draw counters. Used by paceAndRecordFrame() and unit tests.
    void recordGeneratedFrameDurationNs(int64_t durationNs) noexcept;

    // Module 13: how many frames reach the display for the real frame about to be recorded.
    // 0 = Frame Blend inactive (legacy pacing), 1 = real only, 2 = generated + real.
    void setPresentsThisFrame(int presents) noexcept {
        mPresentsThisFrame.store(presents, std::memory_order_relaxed);
    }

    void resetCounters() noexcept;

    FrameSnapshot getSnapshot() const noexcept;
    GovernorTier getCurrentTier() const noexcept {
        return static_cast<GovernorTier>(mGovernorTier.load(std::memory_order_relaxed));
    }

    int getLastFrameDrawCalls() const noexcept {

// ============================================================================
// Module 13: Frame Blend (Experimental 1.5x / 2.0x frame generation)
//
// Inspired by "Mine Frame Gen" (Fabric). It is NOT motion-vector frame
// generation: the generated frame is a weighted blend of the previous and the
// current rendered frame (mix(current, previous, strength)). It is presented
// right before the real frame, so it costs roughly one display-frame of extra
// latency and can show ghosting on fast motion.
//
// How it works on LeviLaunchroid (Android GL ES):
//   1. Before eglSwapBuffers the finished back buffer is copied to a texture.
//   2. On "generation" frames a full-screen shader blends previous + current
//      into the back buffer and presents it (extra eglSwapBuffers).
//   3. The real frame is redrawn from its texture and presented normally.
// While active it forces EGL swap interval 1 (VSync) so each queued frame is
// shown on its own vsync. It only helps on displays whose refresh rate is at
// least (real FPS x multiplier), e.g. 90/120 Hz screens.
// ============================================================================

namespace FrameBlendConsts {
// Real frames slower than this (< 20 FPS) are never blended: frames that far
// apart would only look like ghosting.
inline constexpr int64_t kMaxRealFrameNsForGeneration = 50'000'000LL;
inline constexpr int     kModeX15 = 0; // 1 generated frame every 2 rendered frames
inline constexpr int     kModeX20 = 1; // 1 generated frame per rendered frame
inline constexpr int     kMinStrengthPct = 10;
inline constexpr int     kMaxStrengthPct = 90;
} // namespace FrameBlendConsts

struct FrameBlendPlan {
    bool generate{false};      // present a blended frame before the real frame
    int  presentsThisFrame{1}; // 1 (real only) or 2 (generated + real)
};

// Pure scheduling logic (no GL, fully host-testable).
class FrameBlendPlanner {
public:
    // 0 -> 1.5, 1 (or anything else) -> 2.0
    static float multiplierForMode(int mode) noexcept;

    // Blend weight of the PREVIOUS frame in the generated frame (0.1 .. 0.9).
    static float weightForStrengthPct(int strengthPct) noexcept;

    // Decide what to do for the real frame that just finished rendering.
    //  - realFrameNs: time since the previous real frame was handed to the swap
    //  - havePrevious: a previous frame texture exists
    FrameBlendPlan planFrame(int mode, int64_t realFrameNs, bool havePrevious) noexcept;

    void reset() noexcept { mFrameCounter = 0; }

private:
    uint64_t mFrameCounter{0};
};

// Callbacks supplied by NativeHooks so the GL engine can present frames
// without knowing about the hook plumbing. EGL handles are passed as void*.
struct FrameBlendPresentOps {
    bool (*present)(void* eglDisplay, void* eglSurface){nullptr}; // original eglSwapBuffers
    void (*beforePresent)(){nullptr};                             // e.g. TBDR depth/stencil discard
};

uint32_t parseHexColorArgb(std::string_view hex, uint32_t fallback) noexcept {
    if (hex.empty()) return fallback;
    if (hex.front() == '#') hex.remove_prefix(1);
    if (hex.size() != 6 && hex.size() != 8) return fallback;

    uint32_t value = 0;
    auto [ptr, ec] = std::from_chars(hex.data(), hex.data() + hex.size(), value, 16);
    if (ec != std::errc{} || ptr != hex.data() + hex.size()) {
        return fallback;
    }
    if (hex.size() == 6) {
        return 0xFF000000u | value;
    }
    return value;
}

std::string formatHexColorRgb(uint32_t argb) {
    char buf[16]{};
    std::snprintf(buf, sizeof(buf), "#%06X", static_cast<unsigned int>(argb & 0x00FFFFFFu));
    return std::string(buf);
}

namespace {

template <typename T>
void readJsonAtom(const nlohmann::json& j, const char* key, std::atomic<T>& target, T minVal, T maxVal) noexcept {
    if (!j.is_object() || !j.contains(key)) return;
    try {
        T val = j.at(key).get<T>();
        if (val < minVal) val = minVal;
        if (val > maxVal) val = maxVal;
        target.store(val, std::memory_order_relaxed);
    } catch (...) {
    }
}

void readJsonBool(const nlohmann::json& j, const char* key, std::atomic<bool>& target) noexcept {
    if (!j.is_object() || !j.contains(key)) return;
    try {
        if (j.at(key).is_boolean()) {
            target.store(j.at(key).get<bool>(), std::memory_order_relaxed);
        }
    } catch (...) {
    }
}

bool loadConfigFromDisk(const std::filesystem::path& configDir) noexcept {
    try {
        const auto cfgPath = configDir / "config.json";
        if (!std::filesystem::exists(cfgPath)) {
            saveConfigToDisk(configDir);
            writeConfigSchemaToDisk(configDir);
            return true;
        }
        std::ifstream ifs(cfgPath);
        if (!ifs) return false;

        nlohmann::json root = nlohmann::json::parse(ifs, nullptr, false);
        if (root.is_discarded() || !root.is_object()) return false;

        auto& c = getConfig();

        if (root.contains("frame_pacer")) {
            const auto& s = root["frame_pacer"];
            readJsonBool(s, "enabled", c.modFramePacerEnabled);
            readJsonBool(s, "vsync_bypass", c.vsyncBypass);
            readJsonAtom<int>(s, "target_fps", c.targetFps, 60, 240);
            readJsonAtom<int>(s, "pacing_mode", c.pacingMode, 0, 2);
            readJsonBool(s, "anti_stutter_guard", c.antiStutterGuard);
        }
        if (root.contains("engine_options")) {
            const auto& s = root["engine_options"];
            readJsonBool(s, "enabled", c.modEngineOptionsEnabled);
            readJsonBool(s, "disable_fancy_gfx", c.disableFancyGfx);
            readJsonBool(s, "opaque_leaves", c.opaqueLeaves);
            readJsonBool(s, "fast_skies_clouds", c.fastSkiesClouds);
            readJsonBool(s, "disable_msaa_texel_aa", c.disableMsaaTexelAa);
            readJsonBool(s, "fast_lighting", c.fastLighting);
            readJsonAtom<int>(s, "view_distance_chunks", c.viewDistanceChunks, 5, 16);
        }
        if (root.contains("cpu_scheduler")) {
            const auto& s = root["cpu_scheduler"];
            readJsonBool(s, "enabled", c.modCpuSchedulerEnabled);
            readJsonBool(s, "pin_render_thread", c.pinRenderThread);
            readJsonBool(s, "boost_thread_priority", c.boostThreadPriority);
            readJsonAtom<int>(s, "core_selection_mode", c.coreSelectionMode, 0, 2);

inline void cpuRelaxYield() noexcept {
#if defined(__aarch64__)
    asm volatile("yield" ::: "memory");
#elif defined(__x86_64__) || defined(__i386__)
    asm volatile("pause" ::: "memory");
#endif
}

}

int64_t TelemetryEngine::monotonicNowNs() noexcept {
    struct timespec ts {};
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return static_cast<int64_t>(ts.tv_sec) * 1'000'000'000LL + static_cast<int64_t>(ts.tv_nsec);
}

void TelemetryEngine::paceAndRecordFrame() noexcept {
    const auto& cfg = getConfig();
    int64_t nowNs = monotonicNowNs();

    if (mLastSwapTimestampNs == 0) {
        mLastSwapTimestampNs = nowNs;
        return;
    }

    int64_t rawElapsedNs = nowNs - mLastSwapTimestampNs;
    if (rawElapsedNs < 100'000LL) {
        rawElapsedNs = 100'000LL;
    }

    // --- Module 13: Frame Blend. 0 = inactive (legacy behaviour), 1 = real only, 2 = generated + real.
    // While active the swap is VSync-locked, so the sleep pacer stays idle and every presented
    // frame (generated ones included) counts towards the displayed FPS.
    const int presents = std::clamp(mPresentsThisFrame.exchange(0, std::memory_order_relaxed), 0, 2);
    const bool blendLocked = presents >= 1;

    // --- Module 1: 90 FPS Unlocker & Hybrid Micro-Pacer ---
    const bool pacerActive =
        cfg.modFramePacerEnabled.load(std::memory_order_relaxed) && !blendLocked;
    const int pacingMode = cfg.pacingMode.load(std::memory_order_relaxed);
    const int targetFps = std::clamp(cfg.targetFps.load(std::memory_order_relaxed), 60, 240);

    // Target frame budget in nanoseconds (e.g. 90 FPS -> 11,111,111 ns)
    const int64_t targetFrameNs = 1'000'000'000LL / static_cast<int64_t>(targetFps);

    // Anti-Stutter Spike Guard: if the previous frame exceeded budget by >5%,
    // skip pacing sleep on this frame so the render queue catches up immediately.
    const bool skipWaitDueToSpike =
        cfg.antiStutterGuard.load(std::memory_order_relaxed) && mLastFrameWasSpike;

    if (pacerActive && pacingMode > 0 && !skipWaitDueToSpike && rawElapsedNs < targetFrameNs) {
        const int64_t deadlineNs = mLastSwapTimestampNs + targetFrameNs;
        int64_t remainingNs = deadlineNs - nowNs;

        // Coarse kernel sleep up to 350 us before deadline
        constexpr int64_t kSpinMarginNs = 350'000LL;
        if (remainingNs > kSpinMarginNs) {
            const int64_t sleepNs = remainingNs - kSpinMarginNs;
            struct timespec req {};
            req.tv_sec = static_cast<time_t>(sleepNs / 1'000'000'000LL);
            req.tv_nsec = static_cast<long>(sleepNs % 1'000'000'000LL);
            clock_nanosleep(CLOCK_MONOTONIC, 0, &req, nullptr);
            nowNs = monotonicNowNs();
        }

        // Hybrid Micro-Pace sub-350us spin-yield to hit exact 11.111ms frame cadence
        if (pacingMode == 2) {
            int spinGuard = 0;
            while (nowNs < deadlineNs && spinGuard < 4096) {
                cpuRelaxYield();
                nowNs = monotonicNowNs();
                ++spinGuard;
            }
        }
    }

    const int64_t finalDurationNs = std::max<int64_t>(100'000LL, nowNs - mLastSwapTimestampNs);
    mLastSwapTimestampNs = nowNs;

    if (blendLocked) {
        // Split the real frame time evenly across every frame that reached the display.
        const int64_t shareNs = std::max<int64_t>(100'000LL, finalDurationNs / presents);
        mLastFrameWasSpike = (shareNs > (targetFrameNs + targetFrameNs / 18));
        for (int i = 1; i < presents; ++i) {
            recordGeneratedFrameDurationNs(shareNs);
        }
        recordFrameDurationNs(shareNs);
        return;
    }

    mLastFrameWasSpike = (finalDurationNs > (targetFrameNs + targetFrameNs / 18));
    recordFrameDurationNs(finalDurationNs);
}

void TelemetryEngine::recordFrameDurationNs(int64_t durationNs) noexcept {
    durationNs = std::clamp<int64_t>(durationNs, 100'000LL, 500'000'000LL);

    const int draws = mCurrentFrameDraws.exchange(0, std::memory_order_relaxed);
    const int culled = mCurrentFrameCulled.exchange(0, std::memory_order_relaxed);
    mLastFrameDraws.store(draws, std::memory_order_relaxed);
    mLastFrameCulled.store(culled, std::memory_order_relaxed);

    const uint64_t frameNo = mTotalFrames.fetch_add(1, std::memory_order_relaxed) + 1;
    pushSampleAndRefresh(durationNs, frameNo);
}

void TelemetryEngine::recordGeneratedFrameDurationNs(int64_t durationNs) noexcept {
    durationNs = std::clamp<int64_t>(durationNs, 100'000LL, 500'000'000LL);
    mGeneratedFrames.fetch_add(1, std::memory_order_relaxed);
    pushSampleAndRefresh(durationNs, mTotalFrames.load(std::memory_order_relaxed));
}

void TelemetryEngine::pushSampleAndRefresh(int64_t durationNs, uint64_t frameNo) noexcept {
    mFrameDurationsNs[mWriteIndex] = durationNs;
    mWriteIndex = (mWriteIndex + 1) % kWindowSize;
    if (mSampleCount < kWindowSize) {
        ++mSampleCount;
    }

    const float lastMs = static_cast<float>(durationNs) / 1'000'000.0f;
    mLastFrameMs.store(lastMs, std::memory_order_relaxed);

    // Recompute windowed average FPS, 1% Low FPS, and Jitter every frame (window <= 120)
    int64_t totalNs = 0;
    int64_t worstNs = 0;
    int64_t secondWorstNs = 0;
    int64_t jitterAccumNs = 0;

    int64_t prevNs = mFrameDurationsNs[0];
    for (size_t i = 0; i < mSampleCount; ++i) {
        const int64_t d = mFrameDurationsNs[i];
        totalNs += d;
        if (d >= worstNs) {
            secondWorstNs = worstNs;
            worstNs = d;
        } else if (d > secondWorstNs) {
            secondWorstNs = d;
        }
        if (i > 0) {
            jitterAccumNs += std::abs(d - prevNs);
            prevNs = d;
        }
    }

    const float avgNs = static_cast<float>(totalNs) / static_cast<float>(mSampleCount);
    const float avgFps = (avgNs > 1.0f) ? (1'000'000'000.0f / avgNs) : 90.0f;
    const int64_t lowRefNs = (mSampleCount >= 20 && secondWorstNs > 0) ? secondWorstNs : worstNs;
    const float onePctLowFps = (lowRefNs > 0)
        ? (1'000'000'000.0f / static_cast<float>(lowRefNs))
        : avgFps;
    const float jitterMs = (mSampleCount > 1)
        ? (static_cast<float>(jitterAccumNs) / static_cast<float>(mSampleCount - 1)) / 1'000'000.0f
        : 0.0f;

    mAvgFps.store(avgFps, std::memory_order_relaxed);
    mOnePercentLowFps.store(onePctLowFps, std::memory_order_relaxed);
    mJitterMs.store(jitterMs, std::memory_order_relaxed);

    // Evaluate the 4-tier Adaptive 90-Lock Governor
    if ((frameNo & 0x7u) == 0u || mSampleCount <= 16) {
        evaluateGovernor(avgFps, onePctLowFps, lastMs);
    }
}

void TelemetryEngine::evaluateGovernor(float avgFps, float onePctLow, float lastFrameMs) noexcept {
    const auto& cfg = getConfig();
    if (!cfg.modAdaptiveGovernorEnabled.load(std::memory_order_relaxed)) {
        mGovernorTier.store(static_cast<int>(GovernorTier::Locked90Plus), std::memory_order_relaxed);
        return;
    }

    if (cfg.ultraBoostOverride.load(std::memory_order_relaxed)) {
        mGovernorTier.store(static_cast<int>(GovernorTier::Ultra90Lock), std::memory_order_relaxed);
        return;
    }

    if (!cfg.adaptive90Governor.load(std::memory_order_relaxed)) {
        mGovernorTier.store(static_cast<int>(GovernorTier::Balanced90), std::memory_order_relaxed);
        return;
    }

    const float target = static_cast<float>(cfg.targetFps.load(std::memory_order_relaxed));
    GovernorTier nextTier = GovernorTier::Locked90Plus;

    if (avgFps < target * 0.86f || onePctLow < target * 0.78f || lastFrameMs > 14.5f) {
        nextTier = GovernorTier::Ultra90Lock;
    } else if (avgFps < target * 0.94f || onePctLow < target * 0.86f) {
        nextTier = GovernorTier::Aggressive90;
    } else if (avgFps < target * 0.99f || onePctLow < target * 0.92f) {
        nextTier = GovernorTier::Balanced90;
    } else {
        nextTier = GovernorTier::Locked90Plus;
    }

    mGovernorTier.store(static_cast<int>(nextTier), std::memory_order_relaxed);
}

    mGeneratedFrames.store(0, std::memory_order_relaxed);
    mPresentsThisFrame.store(0, std::memory_order_relaxed);
    mTotalFrames.store(0, std::memory_order_relaxed);

    mWriteIndex = 0;
    mLastSwapTimestampNs = 0;
}

    s.lastFrameMs             = mLastFrameMs.load(std::memory_order_relaxed);
    s.jitterMs                = mJitterMs.load(std::memory_order_relaxed);

std::string EngineOptionsTuner::tuneOptionsContent(std::string_view originalContent) {
    const auto desiredList = buildDesiredOptionPairs();
    std::unordered_map<std::string, std::string> desiredMap;
    desiredMap.reserve(desiredList.size());
    for (const auto& [k, v] : desiredList) {
        desiredMap[k] = v;
    }

    std::unordered_set<std::string> seenKeys;
    std::ostringstream out;

    std::string contentStr(originalContent);
    std::istringstream in(contentStr);
    std::string line;

    while (std::getline(in, line)) {
        if (!line.empty() && line.back() == '\r') {
            line.pop_back();
        }
        const auto colonPos = line.find(':');
        if (colonPos != std::string::npos && colonPos > 0) {
            std::string key = line.substr(0, colonPos);
            auto it = desiredMap.find(key);
            if (it != desiredMap.end()) {
                out << key << ':' << it->second << '\n';
                seenKeys.insert(key);
                continue;
            }
        }
        out << line << '\n';
    }

    // Append any desired keys not yet present in options.txt
    for (const auto& [k, v] : desiredList) {
        if (seenKeys.find(k) == seenKeys.end()) {
            out << k << ':' << v << '\n';
            seenKeys.insert(k);
        }
    }

    return out.str();
}

bool EngineOptionsTuner::tuneOptionsFile(const std::filesystem::path& optionsFilePath) noexcept {
    try {
        if (!std::filesystem::exists(optionsFilePath)) {
            return false;
        }

        // Create backup once if not already present
        std::filesystem::path backupPath = optionsFilePath;
        backupPath += ".hyper90.bak";
        if (!std::filesystem::exists(backupPath)) {
            std::error_code ec;
            std::filesystem::copy_file(
                optionsFilePath, backupPath, std::filesystem::copy_options::skip_existing, ec);
        }

        std::ifstream ifs(optionsFilePath, std::ios::in | std::ios::binary);
        if (!ifs) return false;
        std::ostringstream ss;
        ss << ifs.rdbuf();
        ifs.close();

        const std::string original = ss.str();
        const std::string tuned = tuneOptionsContent(original);

        if (tuned == original) {
            return true;
        }

        std::ofstream ofs(optionsFilePath, std::ios::out | std::ios::trunc | std::ios::binary);
        if (!ofs) return false;
        ofs.write(tuned.data(), static_cast<std::streamsize>(tuned.size()));
        ofs.flush();
        return ofs.good();
    } catch (...) {
        return false;
    }
}

int EngineOptionsTuner::discoverAndTuneAll(const std::filesystem::path& modDir) noexcept {
    if (!getConfig().modEngineOptionsEnabled.load(std::memory_order_relaxed)) {
        return 0;
    }

    int tunedCount = 0;
    std::unordered_set<std::string> visited;

    for (const auto& path : collectCandidatePaths(modDir)) {
        try {
            const std::string key = path.lexically_normal().string();
            if (!visited.insert(key).second) continue;

            if (std::filesystem::exists(path)) {
                if (tuneOptionsFile(path)) {
                    ++tunedCount;
                }
            }
        } catch (...) {
        }
    }

    if (tunedCount > 0) {
        getTelemetry().setOptionsFilesTuned(tunedCount);
    }
    return tunedCount;
}

int EngineOptionsTuner::restoreAllBackups(const std::filesystem::path& modDir) noexcept {
    int restored = 0;
    for (const auto& path : collectCandidatePaths(modDir)) {
        try {
            std::filesystem::path backupPath = path;
            backupPath += ".hyper90.bak";
            if (std::filesystem::exists(backupPath)) {
                std::error_code ec;
                std::filesystem::copy_file(
                    backupPath, path, std::filesystem::copy_options::overwrite_existing, ec);
                if (!ec) {
                    ++restored;
                }
            }
        } catch (...) {
        }
    }
    return restored;
}

std::vector<int> CpuSchedulerOptimizer::selectPerformanceCores(
    const std::vector<int64_t>& coreMaxFreqKhz, int mode) {
    const int n = static_cast<int>(coreMaxFreqKhz.size());
    std::vector<int> selected;
    if (n <= 0) return selected;

    if (mode == 2 || n <= 2) {
        for (int i = 0; i < n; ++i) selected.push_back(i);
        return selected;
    }

    int64_t maxFreq = 0;
    int64_t minFreq = INT64_MAX;
    for (int64_t f : coreMaxFreqKhz) {
        if (f > maxFreq) maxFreq = f;
        if (f > 0 && f < minFreq) minFreq = f;
    }

    // If mode == 0 (AutoMaxFreq) and sysfs reported heterogeneous frequencies:
    // select all cores running at >= 75% of the prime core's max frequency.
    if (mode == 0 && maxFreq > 0 && minFreq < maxFreq) {
        const int64_t threshold = (maxFreq * 75) / 100;
        for (int i = 0; i < n; ++i) {
            if (coreMaxFreqKhz[i] >= threshold) {
                selected.push_back(i);
            }
        }
        if (!selected.empty()) {
            return selected;
        }
    }

    // Fallback or mode == 1 (UpperHalf): on standard Android big.LITTLE SoCs,
    // cores [n/2 .. n-1] are the performance (Big + Prime) cluster.
    const int startCore = std::max(1, n / 2);
    for (int i = startCore; i < n; ++i) {
        selected.push_back(i);
    }
    return selected;
}

int64_t readSysfsCpuMaxFreqKhz(int coreIndex) noexcept {
    char pathBuf[128]{};
    std::snprintf(
        pathBuf, sizeof(pathBuf),
        "/sys/devices/system/cpu/cpu%d/cpufreq/cpuinfo_max_freq", coreIndex);
    std::ifstream ifs(pathBuf);
    if (!ifs) {
        std::snprintf(
            pathBuf, sizeof(pathBuf),
            "/sys/devices/system/cpu/cpu%d/cpufreq/scaling_max_freq", coreIndex);
        ifs.open(pathBuf);
    }
    int64_t freq = 0;
    if (ifs) {
        ifs >> freq;
    }
    return freq;
}

int CpuSchedulerOptimizer::applyRenderThreadBoost(bool /*forceReapply*/) noexcept {
    const auto& cfg = getConfig();
    if (!cfg.modCpuSchedulerEnabled.load(std::memory_order_relaxed)) {
        return 0;
    }

    long numCores = sysconf(_SC_NPROCESSORS_CONF);
    if (numCores <= 0 || numCores > 32) {
        numCores = 8;
    }

    std::vector<int64_t> freqs(static_cast<size_t>(numCores), 0);
    for (int i = 0; i < static_cast<int>(numCores); ++i) {
        freqs[static_cast<size_t>(i)] = readSysfsCpuMaxFreqKhz(i);
    }

    const int mode = cfg.coreSelectionMode.load(std::memory_order_relaxed);
    const auto selectedCores = selectPerformanceCores(freqs, mode);

    if (cfg.pinRenderThread.load(std::memory_order_relaxed) && !selectedCores.empty()) {
        cpu_set_t mask;
        CPU_ZERO(&mask);
        for (int core : selectedCores) {
            if (core >= 0 && core < CPU_SETSIZE) {
                CPU_SET(core, &mask);
            }
        }
        (void)sched_setaffinity(0, sizeof(cpu_set_t), &mask);
    }

    if (cfg.boostThreadPriority.load(std::memory_order_relaxed)) {
        // Reduce timer slack to 10 microseconds so clock_nanosleep in Frame Pacer wakes accurately
#if defined(PR_SET_TIMERSLACK)
        (void)prctl(PR_SET_TIMERSLACK, 10'000UL, 0UL, 0UL, 0UL);
#endif
        // Try high-priority nice values (-10 -> -8 -> -5 -> -2)
        for (int niceVal : {-10, -8, -5, -2}) {
            if (setpriority(PRIO_PROCESS, 0, niceVal) == 0) {
                break;
            }
        }
    }

    const int pinnedCount = static_cast<int>(selectedCores.size());
    getTelemetry().setPinnedBigCores(pinnedCount);
    return pinnedCount;
}

void CpuSchedulerOptimizer::onRenderFrameTick() noexcept {
    static int64_t sLastBoostNs = 0;
    const auto& cfg = getConfig();
    if (!cfg.modCpuSchedulerEnabled.load(std::memory_order_relaxed)) {
        return;
    }

    const int64_t nowNs = TelemetryEngine::monotonicNowNs();
    const int64_t intervalNs =
        static_cast<int64_t>(std::clamp(cfg.reapplyIntervalSec.load(std::memory_order_relaxed), 1, 30)) *
        1'000'000'000LL;

    if (sLastBoostNs == 0 || (nowNs - sLastBoostNs) >= intervalNs) {
        sLastBoostNs = nowNs;
        applyRenderThreadBoost(false);
    }
}

bool MemoryArenaOptimizer::applyBionicHeapTuning() noexcept {

bool MemoryArenaOptimizer::lockHotMemoryPages() noexcept {
    const auto& cfg = getConfig();
    if (!cfg.modMemoryArenaEnabled.load(std::memory_order_relaxed) ||
        !cfg.lockHotPages.load(std::memory_order_relaxed)) {
        return false;

    // Touch every page in the hot scratch arena to fault it in upfront
    for (size_t offset = 0; offset < sizeof(sHotScratchArena); offset += 4096) {
        sHotScratchArena[offset] = static_cast<uint8_t>(offset & 0xFFu);

        (void)munlock(&getHudCache(), sizeof(HudCacheEngine));

void CameraInput90HzStabilizer::filterTurnDelta(float& dx, float& dy, float dtSeconds) noexcept {
    const auto& cfg = getConfig();
    if (!cfg.modCameraInputEnabled.load(std::memory_order_relaxed) ||
        !cfg.smoothCamera90Hz.load(std::memory_order_relaxed)) {
        return;
    }

    if (!std::isfinite(dx) || !std::isfinite(dy)) {
        dx = 0.0f;
        dy = 0.0f;
        return;
    }

    // Raw Flick Bypass: instant 180-degree turns pass through with zero filter lag
    if (cfg.rawFlickBypass.load(std::memory_order_relaxed)) {
        const float magSq = dx * dx + dy * dy;
        if (magSq > 900.0f) { // > 30 degrees/frame flick
            mSmoothedDx = dx;
            mSmoothedDy = dy;
            mInitialized = true;
            return;
        }
    }

    dtSeconds = std::clamp(dtSeconds, 0.002f, 0.050f);
    const float resp = std::clamp(cfg.cameraResponsiveness.load(std::memory_order_relaxed), 10.0f, 60.0f);
    const float alpha = std::clamp(1.0f - std::exp(-resp * dtSeconds), 0.15f, 1.0f);

    if (!mInitialized) {
        mSmoothedDx = dx;
        mSmoothedDy = dy;
        mInitialized = true;
        return;
    }

    mSmoothedDx += (dx - mSmoothedDx) * alpha;
    mSmoothedDy += (dy - mSmoothedDy) * alpha;
    dx = mSmoothedDx;
    dy = mSmoothedDy;
}

void CameraInput90HzStabilizer::reset() noexcept {
    mSmoothedDx = 0.0f;
    mSmoothedDy = 0.0f;
    mInitialized = false;
}

void CameraInput90HzStabilizer::optimizeSocketDescriptor(int fd) noexcept {
    if (fd < 0) return;
    const auto& cfg = getConfig();
    if (!cfg.modCameraInputEnabled.load(std::memory_order_relaxed) ||
        !cfg.lowDelaySockets.load(std::memory_order_relaxed)) {
        return;
    }

    int one = 1;
    (void)setsockopt(fd, IPPROTO_TCP, TCP_NODELAY, &one, sizeof(one));
    int tos = 0x10; // IPTOS_LOWDELAY
    (void)setsockopt(fd, IPPROTO_IP, IP_TOS, &tos, sizeof(tos));
    int prio = 6;
    (void)setsockopt(fd, SOL_SOCKET, SO_PRIORITY, &prio, sizeof(prio));
}

bool GlPipelineOptimizer::shouldSkipCapabilityChange(uint32_t cap, bool enable) noexcept {
    const auto& cfg = getConfig();
    if (!cfg.modGlPipelineEnabled.load(std::memory_order_relaxed)) {
        return false;
    }

    // Block costly mobile fragment capabilities when enabled
    if (enable && cfg.disableDitherMsaa.load(std::memory_order_relaxed)) {
        if (cap == GlConsts::kGlDither ||
            cap == GlConsts::kGlMultisampleExt ||
            cap == GlConsts::kGlSampleAlphaToCoverage) {
            getTelemetry().onGlStateDeduped();
            return true;
        }
    }

    if (!cfg.dedupGlState.load(std::memory_order_relaxed)) {
        return false;
    }

    const int8_t desired = enable ? 1 : 0;
    int8_t* tracked = nullptr;
    switch (cap) {
        case GlConsts::kGlBlend:       tracked = &mBlendState; break;
        case GlConsts::kGlDepthTest:   tracked = &mDepthTestState; break;
        case GlConsts::kGlCullFace:    tracked = &mCullFaceState; break;
        case GlConsts::kGlScissorTest: tracked = &mScissorTestState; break;
        default: break;
    }

    if (tracked != nullptr) {
        if (*tracked == desired) {
            getTelemetry().onGlStateDeduped();
            return true;
        }
        *tracked = desired;
    }
    return false;
}

uint32_t GlPipelineOptimizer::rewriteHintMode(uint32_t target, uint32_t mode) const noexcept {
    const auto& cfg = getConfig();
    if (!cfg.modGlPipelineEnabled.load(std::memory_order_relaxed) ||
        !cfg.fastestShaderHints.load(std::memory_order_relaxed)) {
        return mode;
    }

    if (target == GlConsts::kGlGenerateMipmapHint ||
        target == GlConsts::kGlFragmentShaderDerivativeHint) {
        return GlConsts::kGlFastest;
    }
    return mode;
}

bool GlPipelineOptimizer::shouldSkipUseProgram(uint32_t program) noexcept {
    const auto& cfg = getConfig();
    if (!cfg.modGlPipelineEnabled.load(std::memory_order_relaxed) ||
        !cfg.dedupGlState.load(std::memory_order_relaxed)) {
        mCurrentProgram = program;
        return false;
    }

    if (program != 0 && mCurrentProgram == program) {
        getTelemetry().onGlStateDeduped();
        return true;
    }
    mCurrentProgram = program;
    return false;
}

void GlPipelineOptimizer::onActiveTexture(uint32_t textureEnum) noexcept {
    if (textureEnum >= GlConsts::kGlTexture0 &&
        textureEnum < GlConsts::kGlTexture0 + kTrackedTexUnits) {
        mActiveTexUnit = textureEnum - GlConsts::kGlTexture0;
    } else {
        mActiveTexUnit = 0;
    }
}

bool GlPipelineOptimizer::shouldSkipBindTexture(uint32_t target, uint32_t texture) noexcept {
    if (target != GlConsts::kGlTexture2D || mActiveTexUnit >= kTrackedTexUnits) {
        return false;
    }

    const auto& cfg = getConfig();
    if (!cfg.modGlPipelineEnabled.load(std::memory_order_relaxed) ||
        !cfg.dedupGlState.load(std::memory_order_relaxed)) {
        mBoundTex2D[mActiveTexUnit] = texture;
        return false;
    }

    if (texture != 0 && mBoundTex2D[mActiveTexUnit] == texture) {
        getTelemetry().onGlStateDeduped();
        return true;
    }
    mBoundTex2D[mActiveTexUnit] = texture;
    return false;
}

bool GlPipelineOptimizer::shouldSkipDepthMask(uint8_t flag) noexcept {
    const auto& cfg = getConfig();
    const int8_t desired = (flag != 0) ? 1 : 0;
    if (!cfg.modGlPipelineEnabled.load(std::memory_order_relaxed) ||
        !cfg.dedupGlState.load(std::memory_order_relaxed)) {
        mDepthMaskState = desired;
        return false;
    }

    if (mDepthMaskState == desired) {
        getTelemetry().onGlStateDeduped();
        return true;
    }
    mDepthMaskState = desired;
    return false;
}

uint32_t GlPipelineOptimizer::rewriteClearMask(uint32_t mask) const noexcept {
    const auto& cfg = getConfig();
    if (!cfg.modTbdrTileSaverEnabled.load(std::memory_order_relaxed) ||
        !cfg.optimizeClearCalls.load(std::memory_order_relaxed)) {
        return mask;
    }

    // On packed D24S8 TBDR tile buffers, clearing depth without stencil forces
    // the GPU to load the previous frame's stencil tile from system RAM.
    if ((mask & GlConsts::kGlDepthBufferBit) != 0u) {
        mask |= GlConsts::kGlStencilBufferBit;
    }
    return mask;
}

void GlPipelineOptimizer::rewriteTexParameteri(uint32_t target, uint32_t pname, int32_t& param) noexcept {
    if (target != GlConsts::kGlTexture2D) return;
    const auto& cfg = getConfig();
    if (!cfg.modTextureLodEnabled.load(std::memory_order_relaxed)) return;

    const auto tier = getTelemetry().getCurrentTier();

    if (pname == GlConsts::kGlTextureMinFilter && cfg.fastMipmapFilter.load(std::memory_order_relaxed)) {
        if (static_cast<uint32_t>(param) == GlConsts::kGlLinearMipmapLinear) {
            param = static_cast<int32_t>(
                (tier == GovernorTier::Ultra90Lock)
                    ? GlConsts::kGlNearestMipmapNearest
                    : GlConsts::kGlLinearMipmapNearest);
            getTelemetry().onTextureFilterOptimized();
        } else if (static_cast<uint32_t>(param) == GlConsts::kGlNearestMipmapLinear) {
            param = static_cast<int32_t>(GlConsts::kGlNearestMipmapNearest);
            getTelemetry().onTextureFilterOptimized();
        }
    } else if (pname == GlConsts::kGlTextureMaxAnisotropyExt &&
               cfg.clampAnisotropy.load(std::memory_order_relaxed)) {
        const int maxAniso = (tier >= GovernorTier::Balanced90)
            ? 1
            : std::clamp(cfg.maxAnisotropy.load(std::memory_order_relaxed), 1, 8);
        if (param > maxAniso) {
            param = maxAniso;
            getTelemetry().onTextureFilterOptimized();
        }
    } else if (pname == GlConsts::kGlTextureMaxLevel) {
        const int maxMip = std::clamp(cfg.maxMipLevel.load(std::memory_order_relaxed), 2, 10);
        if (param > maxMip) {
            param = maxMip;
            getTelemetry().onTextureFilterOptimized();
        }
    }
}

void GlPipelineOptimizer::rewriteTexParameterf(uint32_t target, uint32_t pname, float& param) noexcept {
    if (target != GlConsts::kGlTexture2D) return;
    const auto& cfg = getConfig();
    if (!cfg.modTextureLodEnabled.load(std::memory_order_relaxed)) return;

    if (pname == GlConsts::kGlTextureMaxAnisotropyExt &&
        cfg.clampAnisotropy.load(std::memory_order_relaxed)) {
        const auto tier = getTelemetry().getCurrentTier();
        const float maxAniso = (tier >= GovernorTier::Balanced90)
            ? 1.0f
            : static_cast<float>(std::clamp(cfg.maxAnisotropy.load(std::memory_order_relaxed), 1, 8));
        if (param > maxAniso) {
            param = maxAniso;
            getTelemetry().onTextureFilterOptimized();
        }
    } else if (pname == GlConsts::kGlTextureMinFilter &&
               cfg.fastMipmapFilter.load(std::memory_order_relaxed)) {
        int32_t intParam = static_cast<int32_t>(param);
        rewriteTexParameteri(target, pname, intParam);
        param = static_cast<float>(intParam);
    }
}

bool GlPipelineOptimizer::shouldCullDrawCall(uint32_t mode, int32_t count) noexcept {
    const auto& cfg = getConfig();

    // 1. Degenerate draw call rejection
    if (count <= 0) {
        if (cfg.modDrawBatchCullerEnabled.load(std::memory_order_relaxed) &&
            cfg.cullDegenerateDraws.load(std::memory_order_relaxed)) {
            getTelemetry().onDrawCallCulled();
            return true;
        }
        return false;
    }

    const int currentDraws = getTelemetry().getLastFrameDrawCalls();
    const auto tier = getTelemetry().getCurrentTier();

    // 2. Module 7 & 8: Micro-batch & particle budget culling on small triangle batches
    if (mode == GlConsts::kGlTriangles && count <= 36) {
        // Never cull early-frame terrain or late-frame UI if within safe draw budget
        const int maxParticleBudget = std::clamp(
            cfg.maxParticlesPerFrame.load(std::memory_order_relaxed), 20, 1000);
        const int effectiveParticleCap = (tier >= GovernorTier::Aggressive90)
            ? (maxParticleBudget * 60) / 100
            : maxParticleBudget;

        const int maxMicroIndices = std::clamp(
            cfg.microBatchMaxIndices.load(std::memory_order_relaxed), 6, 60);

        if (count <= maxMicroIndices) {
            ++mMicroBatchCounter;

            // Module 8: Hard particle micro-batch cap per frame
            if (cfg.modEntityParticleEnabled.load(std::memory_order_relaxed) &&
                cfg.limitParticles.load(std::memory_order_relaxed) &&
                static_cast<int>(mMicroBatchCounter) > effectiveParticleCap &&
                currentDraws > 64) {
                getTelemetry().onDrawCallCulled();
                return true;
            }

            // Module 7: Adaptive micro-batch decimation when FPS dips below trigger threshold
            if (cfg.modDrawBatchCullerEnabled.load(std::memory_order_relaxed) &&
                cfg.adaptiveMicroBatchCull.load(std::memory_order_relaxed) &&
                currentDraws > 80) {
                const float avgFps = getTelemetry().getSnapshot().avgFps;
                const float triggerFps = static_cast<float>(
                    cfg.cullTriggerFps.load(std::memory_order_relaxed));

                if (tier == GovernorTier::Ultra90Lock || avgFps < triggerFps - 6.0f) {
                    // Cull every 2nd micro-batch in Ultra 90-Lock rescue mode
                    if ((mMicroBatchCounter & 1u) == 1u) {
                        getTelemetry().onDrawCallCulled();
                        return true;
                    }
                } else if (tier == GovernorTier::Aggressive90 || avgFps < triggerFps) {
                    // Cull every 4th micro-batch in Aggressive 90 mode
                    if ((mMicroBatchCounter & 3u) == 3u) {
                        getTelemetry().onDrawCallCulled();
                        return true;
                    }
                }
            }
        }
    }

    getTelemetry().onDrawCallSubmitted();
    return false;
}

// Module 13 callback: presents the current back buffer through the ORIGINAL eglSwapBuffers.
bool presentViaOriginalSwap(void* dpy, void* surface) {
    if (orig_eglSwapBuffers == nullptr) return false;
    return orig_eglSwapBuffers(static_cast<EGLDisplay>(dpy), static_cast<EGLSurface>(surface)) == EGL_TRUE;
}

EGLBoolean hook_eglSwapBuffers(EGLDisplay dpy, EGLSurface surface) {
    tIsRenderThread = true;
    const auto& cfg = getConfig();

    // 1. Enforce the swap interval on the active display:
    //    Module 13 (Frame Blend) -> 1 (VSync), Module 1 (vsync bypass) -> 0,
    //    otherwise whatever the game itself requested.
    if (orig_eglSwapInterval != nullptr && dpy != EGL_NO_DISPLAY) {
        int wantInterval = gEngineSwapInterval.load(std::memory_order_relaxed);
        if (FrameBlendGl::wantsVsync()) {
            wantInterval = 1;
        } else if (cfg.modFramePacerEnabled.load(std::memory_order_relaxed) &&
                   cfg.vsyncBypass.load(std::memory_order_relaxed)) {
            wantInterval = 0;
        }
        if (wantInterval >= 0 && (dpy != gLastDisplay || wantInterval != gAppliedSwapInterval)) {
            gLastDisplay = dpy;
            gAppliedSwapInterval = wantInterval;
            orig_eglSwapInterval(dpy, wantInterval);
        }
    }

    // 2. Late-bind libminecraftpe.so symbols if needed
    if (!gMcSymbolsHooked.load(std::memory_order_relaxed)) {
        NativeHooks::tryResolveMinecraftSymbols();
    }

    // 3. Module 3: Periodic Big.LITTLE Prime Core affinity & priority boost on RenderThread
    CpuSchedulerOptimizer::onRenderFrameTick();

    // 3b. Module 13: Frame Blend. May present a generated frame (extra swap) and redraw the real one.
    FrameBlendPresentOps blendOps{};
    blendOps.present       = &presentViaOriginalSwap;
    blendOps.beforePresent = &discardDepthStencilOnTbdr;
    const int blendPresents = FrameBlendGl::onBeforeSwap(dpy, surface, blendOps);
    getTelemetry().setPresentsThisFrame(blendPresents);

    // 4. Module 5: Discard Depth & Stencil tile writeback before buffer swap
    discardDepthStencilOnTbdr();

    // 5. Reset GL shadow state before external overlays or buffer swap
    getGlOptimizer().resetFrameState();

    // 6. Perform actual buffer swap
    const EGLBoolean res = orig_eglSwapBuffers ? orig_eglSwapBuffers(dpy, surface) : EGL_FALSE;

    // 7. Module 1 & Module 12: Hybrid 90-FPS Micro-Pacer + Telemetry + Adaptive Governor
    getTelemetry().paceAndRecordFrame();

    // 8. Update HUD overlay draw commands
    Modules::onFrameRendered();

    return res;
}

template <typename FnPtr>
void installSymHook(
    void* libHandle,
    const char* symName,
    FnPtr detour,
    FnPtr& origOut,
    pl::memory::HookHandle& handleOut) noexcept {
    if (handleOut.installed() || libHandle == nullptr) return;
    void* target = dlsym(libHandle, symName);
    if (!target) return;
    pl::memory::FuncPtr orig = nullptr;
    handleOut = pl::memory::HookHandle(
        target,
        reinterpret_cast<pl::memory::FuncPtr>(detour),
        &orig,
        pl::memory::HookPriority::Normal);
    if (handleOut.installed() && orig != nullptr) {
        origOut = reinterpret_cast<FnPtr>(orig);
    }
}

void unhookSafe(pl::memory::HookHandle& h) noexcept {
    h.reset();
}

void NativeHooks::tryResolveMinecraftSymbols() noexcept {
    if (gMcSymbolsHooked.load(std::memory_order_relaxed)) return;

    void* mcLib = dlopen("libminecraftpe.so", RTLD_LAZY | RTLD_NOLOAD);
    if (!mcLib) return;

    installSymHook(
        mcLib,
        "_ZN11LocalPlayer9applyTurnEff",
        hook_LocalPlayer_applyTurn,
        orig_LocalPlayer_applyTurn,
        h_LocalPlayer_applyTurn);
    installSymHook(
        mcLib,
        "_ZN11ActorRender19enableInterpolationEP5Actorf",
        hook_ActorRender_enableInterpolation,
        orig_ActorRender_enableInterpolation,
        h_ActorRender_enableInterpolation);
    installSymHook(
        mcLib,
        "_ZNK7Options8getGammaEv",
        hook_Options_getGamma,
        orig_Options_getGamma,
        h_Options_getGamma);

    dlclose(mcLib);
    gMcSymbolsHooked.store(true, std::memory_order_relaxed);
}

bool NativeHooks::installAll() noexcept {
    if (gHooksInstalled.load(std::memory_order_relaxed)) {
        return true;
    }

    // 1. Resolve TBDR framebuffer discard extensions
    void* gles3Lib = dlopen("libGLESv3.so", RTLD_LAZY | RTLD_NOLOAD);
    if (!gles3Lib) gles3Lib = dlopen("libGLESv3.so", RTLD_LAZY);
    if (gles3Lib) {
        gGlInvalidateFramebuffer = reinterpret_cast<PFN_glInvalidateFramebuffer>(
            dlsym(gles3Lib, "glInvalidateFramebuffer"));
    }
    if (!gGlInvalidateFramebuffer) {
        gGlInvalidateFramebuffer = reinterpret_cast<PFN_glInvalidateFramebuffer>(
            eglGetProcAddress("glInvalidateFramebuffer"));
    }
    gGlDiscardFramebufferEXT = reinterpret_cast<PFN_glDiscardFramebufferEXT>(
        eglGetProcAddress("glDiscardFramebufferEXT"));

    // 2. Hook libEGL.so
    void* eglLib = dlopen("libEGL.so", RTLD_LAZY | RTLD_NOLOAD);
    if (!eglLib) eglLib = dlopen("libEGL.so", RTLD_LAZY);
    if (eglLib) {
        installSymHook(eglLib, "eglSwapInterval", hook_eglSwapInterval, orig_eglSwapInterval, h_eglSwapInterval);
        installSymHook(eglLib, "eglSwapBuffers",  hook_eglSwapBuffers,  orig_eglSwapBuffers,  h_eglSwapBuffers);
    }

    // 3. Hook libGLESv2.so
    void* gles2Lib = dlopen("libGLESv2.so", RTLD_LAZY | RTLD_NOLOAD);
    if (!gles2Lib) gles2Lib = dlopen("libGLESv2.so", RTLD_LAZY);
    if (gles2Lib) {
        installSymHook(gles2Lib, "glHint",           hook_glHint,           orig_glHint,           h_glHint);
        installSymHook(gles2Lib, "glEnable",         hook_glEnable,         orig_glEnable,         h_glEnable);
        installSymHook(gles2Lib, "glDisable",        hook_glDisable,        orig_glDisable,        h_glDisable);
        installSymHook(gles2Lib, "glUseProgram",     hook_glUseProgram,     orig_glUseProgram,     h_glUseProgram);
        installSymHook(gles2Lib, "glActiveTexture",  hook_glActiveTexture,  orig_glActiveTexture,  h_glActiveTexture);
        installSymHook(gles2Lib, "glBindTexture",    hook_glBindTexture,    orig_glBindTexture,    h_glBindTexture);
        installSymHook(gles2Lib, "glDepthMask",      hook_glDepthMask,      orig_glDepthMask,      h_glDepthMask);
        installSymHook(gles2Lib, "glFinish",         hook_glFinish,         orig_glFinish,         h_glFinish);
        installSymHook(gles2Lib, "glClear",          hook_glClear,          orig_glClear,          h_glClear);
        installSymHook(gles2Lib, "glTexParameteri",  hook_glTexParameteri,  orig_glTexParameteri,  h_glTexParameteri);
        installSymHook(gles2Lib, "glTexParameterf",  hook_glTexParameterf,  orig_glTexParameterf,  h_glTexParameterf);
        installSymHook(gles2Lib, "glDrawElements",   hook_glDrawElements,   orig_glDrawElements,   h_glDrawElements);
        installSymHook(gles2Lib, "glDrawArrays",     hook_glDrawArrays,     orig_glDrawArrays,     h_glDrawArrays);
    }

    // 4. Hook libc.so (fopen for options.txt auto-tuner, connect for low-delay sockets)
    void* libcHandle = dlopen("libc.so", RTLD_LAZY | RTLD_NOLOAD);
    if (libcHandle) {
        installSymHook(libcHandle, "fopen",   hook_fopen,   orig_fopen,   h_fopen);
        installSymHook(libcHandle, "connect", hook_connect, orig_connect, h_connect);
        dlclose(libcHandle);
    }

    // 5. Try resolving libminecraftpe.so symbols immediately (or defer until first swap)
    tryResolveMinecraftSymbols();

    gHooksInstalled.store(true, std::memory_order_relaxed);
    return true;
}

float FrameBlendPlanner::multiplierForMode(int mode) noexcept {
    return (mode == FrameBlendConsts::kModeX15) ? 1.5f : 2.0f;
}

float FrameBlendPlanner::weightForStrengthPct(int strengthPct) noexcept {
    return static_cast<float>(std::clamp(
               strengthPct, FrameBlendConsts::kMinStrengthPct, FrameBlendConsts::kMaxStrengthPct)) /
           100.0f;
}

FrameBlendPlan FrameBlendPlanner::planFrame(int mode, int64_t realFrameNs, bool havePrevious) noexcept {
    FrameBlendPlan plan{};
    const uint64_t index = mFrameCounter++;

    if (!havePrevious || realFrameNs <= 0 ||
        realFrameNs > FrameBlendConsts::kMaxRealFrameNsForGeneration) {
        return plan;
    }

    const bool everyFrame = (mode != FrameBlendConsts::kModeX15);
    if (everyFrame || (index & 1u) == 0u) {
        plan.generate = true;
        plan.presentsThisFrame = 2;
    }
    return plan;
}

constexpr const char* kVertexShader =
    "#version 300 es\n"
    "out vec2 vUv;\n"
    "void main() {\n"
    "    vec2 p = vec2(float((gl_VertexID << 1) & 2), float(gl_VertexID & 2));\n"
    "    vUv = p;\n"
    "    gl_Position = vec4(p * 2.0 - 1.0, 0.0, 1.0);\n"
    "}\n";

constexpr const char* kFragmentShader =
    "#version 300 es\n"
    "precision mediump float;\n"
    "uniform sampler2D uCur;\n"
    "uniform sampler2D uPrev;\n"
    "uniform float uWeight;\n"
    "in vec2 vUv;\n"
    "out vec4 fragColor;\n"
    "void main() {\n"
    "    vec3 c = texture(uCur, vUv).rgb;\n"
    "    if (uWeight > 0.0) {\n"
    "        c = mix(c, texture(uPrev, vUv).rgb, uWeight);\n"
    "    }\n"
    "    fragColor = vec4(c, 1.0);\n"
    "}\n";

constexpr int kVerifyFrames = 4;

struct Engine {
    EGLContext        ctx{EGL_NO_CONTEXT};
    GLuint            program{0};
    GLuint            vao{0};
    GLuint            tex[2]{0, 0};
    GLint             uCur{-1};
    GLint             uPrev{-1};
    GLint             uWeight{-1};
    int               width{0};
    int               height{0};
    int               curIdx{0};
    bool              havePrev{false};
    int               verifyFramesLeft{kVerifyFrames};
    int64_t           lastRealSwapNs{0};
    FrameBlendPlanner planner{};
};

Engine gEngine;

struct SavedState {
    GLint     program{0};
    GLint     vao{0};
    GLint     activeTex{GL_TEXTURE0};
    GLint     tex[2]{0, 0};
    GLint     sampler[2]{0, 0};
    GLint     drawFbo{0};
    GLint     readFbo{0};
    GLint     readBuffer{GL_BACK};
    GLint     viewport[4]{0, 0, 0, 0};
    GLboolean colorMask[4]{GL_TRUE, GL_TRUE, GL_TRUE, GL_TRUE};
    GLboolean scissor{GL_FALSE};
    GLboolean blend{GL_FALSE};
    GLboolean depth{GL_FALSE};
    GLboolean cull{GL_FALSE};
    GLboolean stencil{GL_FALSE};
};

void setCap(GLenum cap, GLboolean on) noexcept {
    if (on) {
        glEnable(cap);
    } else {
        glDisable(cap);
    }
}

void saveState(SavedState& s) noexcept {
    glGetIntegerv(GL_CURRENT_PROGRAM, &s.program);
    glGetIntegerv(GL_VERTEX_ARRAY_BINDING, &s.vao);
    glGetIntegerv(GL_ACTIVE_TEXTURE, &s.activeTex);
    glGetIntegerv(GL_DRAW_FRAMEBUFFER_BINDING, &s.drawFbo);
    glGetIntegerv(GL_READ_FRAMEBUFFER_BINDING, &s.readFbo);
    glGetIntegerv(GL_VIEWPORT, s.viewport);
    glGetBooleanv(GL_COLOR_WRITEMASK, s.colorMask);
    s.scissor = glIsEnabled(GL_SCISSOR_TEST);
    s.blend   = glIsEnabled(GL_BLEND);
    s.depth   = glIsEnabled(GL_DEPTH_TEST);
    s.cull    = glIsEnabled(GL_CULL_FACE);
    s.stencil = glIsEnabled(GL_STENCIL_TEST);

    for (int unit = 0; unit < 2; ++unit) {
        glActiveTexture(static_cast<GLenum>(GL_TEXTURE0 + unit));
        glGetIntegerv(GL_TEXTURE_BINDING_2D, &s.tex[unit]);
        glGetIntegerv(GL_SAMPLER_BINDING, &s.sampler[unit]);
    }
    glActiveTexture(static_cast<GLenum>(s.activeTex));

    // Read buffer of the default framebuffer (restored explicitly below).
    glBindFramebuffer(GL_READ_FRAMEBUFFER, 0);
    glGetIntegerv(GL_READ_BUFFER, &s.readBuffer);
    glBindFramebuffer(GL_READ_FRAMEBUFFER, static_cast<GLuint>(s.readFbo));
}

void restoreState(const SavedState& s) noexcept {
    glBindFramebuffer(GL_READ_FRAMEBUFFER, 0);
    glReadBuffer(static_cast<GLenum>(s.readBuffer));
    glBindFramebuffer(GL_DRAW_FRAMEBUFFER, static_cast<GLuint>(s.drawFbo));
    glBindFramebuffer(GL_READ_FRAMEBUFFER, static_cast<GLuint>(s.readFbo));

    glViewport(s.viewport[0], s.viewport[1], s.viewport[2], s.viewport[3]);
    glColorMask(s.colorMask[0], s.colorMask[1], s.colorMask[2], s.colorMask[3]);
    setCap(GL_SCISSOR_TEST, s.scissor);
    setCap(GL_BLEND, s.blend);
    setCap(GL_DEPTH_TEST, s.depth);
    setCap(GL_CULL_FACE, s.cull);
    setCap(GL_STENCIL_TEST, s.stencil);

    for (int unit = 0; unit < 2; ++unit) {
        glActiveTexture(static_cast<GLenum>(GL_TEXTURE0 + unit));
        glBindTexture(GL_TEXTURE_2D, static_cast<GLuint>(s.tex[unit]));
        glBindSampler(static_cast<GLuint>(unit), static_cast<GLuint>(s.sampler[unit]));
    }
    glActiveTexture(static_cast<GLenum>(s.activeTex));

    glBindVertexArray(static_cast<GLuint>(s.vao));
    glUseProgram(static_cast<GLuint>(s.program));
}

GLuint compileShader(GLenum type, const char* src) noexcept {
    const GLuint shader = glCreateShader(type);
    if (shader == 0) return 0;
    glShaderSource(shader, 1, &src, nullptr);
    glCompileShader(shader);
    GLint ok = 0;
    glGetShaderiv(shader, GL_COMPILE_STATUS, &ok);
    if (ok == 0) {
        glDeleteShader(shader);
        return 0;
    }
    return shader;
}

bool buildProgram(Engine& e) noexcept {
    const GLuint vs = compileShader(GL_VERTEX_SHADER, kVertexShader);
    const GLuint fs = compileShader(GL_FRAGMENT_SHADER, kFragmentShader);
    if (vs == 0 || fs == 0) {
        if (vs != 0) glDeleteShader(vs);
        if (fs != 0) glDeleteShader(fs);
        return false;
    }

    const GLuint prog = glCreateProgram();
    if (prog == 0) {
        glDeleteShader(vs);
        glDeleteShader(fs);
        return false;
    }
    glAttachShader(prog, vs);
    glAttachShader(prog, fs);
    glLinkProgram(prog);
    glDeleteShader(vs);
    glDeleteShader(fs);

    GLint linked = 0;
    glGetProgramiv(prog, GL_LINK_STATUS, &linked);
    if (linked == 0) {
        glDeleteProgram(prog);
        return false;
    }

    e.program = prog;
    e.uCur    = glGetUniformLocation(prog, "uCur");
    e.uPrev   = glGetUniformLocation(prog, "uPrev");
    e.uWeight = glGetUniformLocation(prog, "uWeight");
    glGenVertexArrays(1, &e.vao);
    return e.vao != 0 && e.uCur >= 0 && e.uPrev >= 0 && e.uWeight >= 0;
}

void deleteTextures(Engine& e, bool contextAlive) noexcept {
    if (contextAlive && (e.tex[0] != 0 || e.tex[1] != 0)) {
        glDeleteTextures(2, e.tex);
    }
    e.tex[0] = 0;
    e.tex[1] = 0;
    e.width = 0;
    e.height = 0;
    e.havePrev = false;
    e.curIdx = 0;
}

void releaseAll(Engine& e, bool contextAlive) noexcept {
    deleteTextures(e, contextAlive);
    if (contextAlive) {
        if (e.program != 0) glDeleteProgram(e.program);
        if (e.vao != 0) glDeleteVertexArrays(1, &e.vao);
    }
    e.program = 0;
    e.vao = 0;
    e.uCur = e.uPrev = e.uWeight = -1;
}

bool createTextures(Engine& e, int w, int h) noexcept {
    glActiveTexture(GL_TEXTURE0);
    glGenTextures(2, e.tex);
    if (e.tex[0] == 0 || e.tex[1] == 0) return false;

    for (int i = 0; i < 2; ++i) {
        glBindTexture(GL_TEXTURE_2D, e.tex[i]);
        glTexStorage2D(GL_TEXTURE_2D, 1, GL_RGB8, w, h);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
    }
    e.width = w;
    e.height = h;
    e.havePrev = false;
    e.curIdx = 0;
    return true;
}

bool ensureResources(Engine& e, int w, int h) noexcept {
    if (e.program == 0 && !buildProgram(e)) {
        return false;
    }
    if (e.tex[0] == 0 || e.width != w || e.height != h) {
        deleteTextures(e, true);
        if (!createTextures(e, w, h)) return false;
    }
    return true;
}

void drawFullscreen(const Engine& e, GLuint prevTex, GLuint curTex, float weight) noexcept {
    glBindFramebuffer(GL_FRAMEBUFFER, 0);
    glViewport(0, 0, e.width, e.height);
    glDisable(GL_SCISSOR_TEST);
    glDisable(GL_BLEND);
    glDisable(GL_DEPTH_TEST);
    glDisable(GL_CULL_FACE);
    glDisable(GL_STENCIL_TEST);
    glColorMask(GL_TRUE, GL_TRUE, GL_TRUE, GL_TRUE);

    // Every pixel is overwritten: tell tile-based GPUs not to load the old color.
    const GLenum colorAttachment = GL_COLOR;
    glInvalidateFramebuffer(GL_FRAMEBUFFER, 1, &colorAttachment);

    glUseProgram(e.program);
    glUniform1i(e.uCur, 0);
    glUniform1i(e.uPrev, 1);
    glUniform1f(e.uWeight, weight);

    glActiveTexture(GL_TEXTURE0);
    glBindSampler(0, 0);
    glBindTexture(GL_TEXTURE_2D, curTex);
    glActiveTexture(GL_TEXTURE1);
    glBindSampler(1, 0);
    glBindTexture(GL_TEXTURE_2D, prevTex);

    glBindVertexArray(e.vao);
    glDrawArrays(GL_TRIANGLES, 0, 3);
}

void failModule(Engine& e, const char* reason) noexcept {
    gFailed.store(true, std::memory_order_relaxed);
    __android_log_print(ANDROID_LOG_WARN, "Hyper90", "Frame Blend disabled: %s", reason);
    releaseAll(e, true);
}

int FrameBlendGl::onBeforeSwap(
    void* eglDisplay, void* eglSurface, const FrameBlendPresentOps& ops) noexcept {
    auto& e = gEngine;
    const auto& cfg = getConfig();
    const EGLContext currentCtx = eglGetCurrentContext();

    // --- Module disabled: free GL objects once, on the thread that owns them ---
    if (!cfg.modFrameBlendEnabled.load(std::memory_order_relaxed)) {
        if (e.program != 0 || e.tex[0] != 0 || e.vao != 0) {
            releaseAll(e, currentCtx != EGL_NO_CONTEXT && currentCtx == e.ctx);
        }
        gFailed.store(false, std::memory_order_relaxed);
        e.havePrev = false;
        e.lastRealSwapNs = 0;
        e.verifyFramesLeft = kVerifyFrames;
        e.planner.reset();
        return 0;
    }

    if (gFailed.load(std::memory_order_relaxed) || ops.present == nullptr ||
        currentCtx == EGL_NO_CONTEXT) {
        return 0;
    }

    // --- New GL context (app resume / surface recreation): old names are dead ---
    if (currentCtx != e.ctx) {
        releaseAll(e, false);
        e.ctx = currentCtx;
        e.lastRealSwapNs = 0;
        e.verifyFramesLeft = kVerifyFrames;
    }
    if (gHistoryResetRequested.exchange(false, std::memory_order_relaxed)) {
        e.havePrev = false;
        e.planner.reset();
    }

    EGLint w = 0;
    EGLint h = 0;
    if (!eglQuerySurface(static_cast<EGLDisplay>(eglDisplay), static_cast<EGLSurface>(eglSurface),
                         EGL_WIDTH, &w) ||
        !eglQuerySurface(static_cast<EGLDisplay>(eglDisplay), static_cast<EGLSurface>(eglSurface),
                         EGL_HEIGHT, &h) ||
        w <= 0 || h <= 0) {
        return 0;
    }

    const int64_t nowNs = TelemetryEngine::monotonicNowNs();
    const int64_t realFrameNs = (e.lastRealSwapNs != 0) ? (nowNs - e.lastRealSwapNs) : 0;
    e.lastRealSwapNs = nowNs;

    // Our own GL calls must not feed the state deduplicator / draw culler.
    GlHookBypassScope bypass;

    SavedState saved;
    saveState(saved);

    if (e.verifyFramesLeft > 0) {
        while (glGetError() != GL_NO_ERROR) {
            // drop errors left behind by the game so verification stays accurate
        }
    }

    if (!ensureResources(e, w, h)) {
        failModule(e, "could not create GL resources (needs OpenGL ES 3.0)");
        restoreState(saved);
        return 0;
    }

    // 1. Capture the finished frame from the default framebuffer.
    glBindFramebuffer(GL_READ_FRAMEBUFFER, 0);
    glReadBuffer(GL_BACK);
    glActiveTexture(GL_TEXTURE0);
    glBindTexture(GL_TEXTURE_2D, e.tex[e.curIdx]);
    glCopyTexSubImage2D(GL_TEXTURE_2D, 0, 0, 0, 0, 0, w, h);

    const GLuint curTex  = e.tex[e.curIdx];
    const GLuint prevTex = e.tex[e.curIdx ^ 1];

    // 2. Generated frame: blend(previous, current) -> present, then redraw the real frame.
    const FrameBlendPlan plan = e.planner.planFrame(
        cfg.frameBlendMode.load(std::memory_order_relaxed), realFrameNs, e.havePrev);

    int presents = 1;
    if (plan.generate) {
        const float weight = FrameBlendPlanner::weightForStrengthPct(
            cfg.frameBlendStrengthPct.load(std::memory_order_relaxed));

        drawFullscreen(e, prevTex, curTex, weight);
        if (ops.beforePresent != nullptr) ops.beforePresent();
        const bool presented = ops.present(eglDisplay, eglSurface);

        // The new back buffer is undefined: redraw the real frame for the caller's swap.
        drawFullscreen(e, curTex, curTex, 0.0f);

        if (presented) {
            presents = 2;
            gGeneratedFrames.fetch_add(1, std::memory_order_relaxed);
        }
    }

    e.curIdx ^= 1;
    e.havePrev = true;

    if (e.verifyFramesLeft > 0) {
        --e.verifyFramesLeft;
        if (glGetError() != GL_NO_ERROR) {
            failModule(e, "GL error while capturing/blending frames");
        }
    }

    restoreState(saved);
    return presents;
}

    char* endPtr = nullptr;
    const float val = std::strtof(tmp.c_str(), &endPtr);
    if (endPtr == tmp.c_str() || !std::isfinite(val)) {
        return fallback;
    }
    return std::clamp(val, minVal, maxVal);
}

void persistConfigQuietly() noexcept {
    if (!gConfigDir.empty()) {
        saveConfigToDisk(gConfigDir);
    }
}

void Modules::handleModuleToggle(std::string_view moduleId, bool enabled) noexcept {
    auto& cfg = getConfig();

    if (moduleId == ModuleIds::kFramePacer) {
        cfg.modFramePacerEnabled.store(enabled, std::memory_order_relaxed);
    } else if (moduleId == ModuleIds::kEngineOptions) {
        cfg.modEngineOptionsEnabled.store(enabled, std::memory_order_relaxed);
        if (enabled && !gModRootDir.empty()) {
            EngineOptionsTuner::discoverAndTuneAll(gModRootDir);
        } else if (!enabled && !gModRootDir.empty()) {
            EngineOptionsTuner::restoreAllBackups(gModRootDir);
        }
    } else if (moduleId == ModuleIds::kCpuScheduler) {
        cfg.modCpuSchedulerEnabled.store(enabled, std::memory_order_relaxed);
        if (enabled) {
            CpuSchedulerOptimizer::applyRenderThreadBoost(true);
        }
    } else if (moduleId == ModuleIds::kGlPipeline) {
        cfg.modGlPipelineEnabled.store(enabled, std::memory_order_relaxed);
        getGlOptimizer().resetFrameState();
    } else if (moduleId == ModuleIds::kTbdrTileSaver) {
        cfg.modTbdrTileSaverEnabled.store(enabled, std::memory_order_relaxed);
    } else if (moduleId == ModuleIds::kTextureLod) {
        cfg.modTextureLodEnabled.store(enabled, std::memory_order_relaxed);
    } else if (moduleId == ModuleIds::kDrawBatchCuller) {
        cfg.modDrawBatchCullerEnabled.store(enabled, std::memory_order_relaxed);
    } else if (moduleId == ModuleIds::kEntityParticleGuard) {
        cfg.modEntityParticleEnabled.store(enabled, std::memory_order_relaxed);
    } else if (moduleId == ModuleIds::kHudUiAccelerator) {
        cfg.modHudAcceleratorEnabled.store(enabled, std::memory_order_relaxed);
        getHudCache().invalidateAll();
    } else if (moduleId == ModuleIds::kMemoryArenaTuner) {
        cfg.modMemoryArenaEnabled.store(enabled, std::memory_order_relaxed);
        if (enabled) {
            MemoryArenaOptimizer::applyBionicHeapTuning();
            MemoryArenaOptimizer::lockHotMemoryPages();
        } else {
            MemoryArenaOptimizer::releaseLockedPages();
        }
    } else if (moduleId == ModuleIds::kCameraInput90Hz) {
        cfg.modCameraInputEnabled.store(enabled, std::memory_order_relaxed);
        getCameraStabilizer().reset();
    } else if (moduleId == ModuleIds::kAdaptiveGovernorHud) {
        cfg.modAdaptiveGovernorEnabled.store(enabled, std::memory_order_relaxed);
    } else if (moduleId == ModuleIds::kFrameBlend) {
        cfg.modFrameBlendEnabled.store(enabled, std::memory_order_relaxed);
        FrameBlendGl::resetHistory();
    }

    persistConfigQuietly();
}

void Modules::unregisterAll() noexcept {
    if (!gRegistered) return;

    pl::modmenu::submitDrawCommands(ModuleIds::kAdaptiveGovernorHud, {});
    pl::modmenu::submitHudEditorElements(ModuleIds::kAdaptiveGovernorHud, {});
    pl::modmenu::submitDrawCommands(ModuleIds::kHudUiAccelerator, {});

    pl::modmenu::unregisterButton(ModuleIds::kBtnPurgeCache);
    pl::modmenu::unregisterButton(ModuleIds::kBtnUltraBoost);

    for (auto it = kModuleDescriptors.rbegin(); it != kModuleDescriptors.rend(); ++it) {
        pl::modmenu::unregisterModule(it->id);
    }

    gRegistered = false;
}

    int fpsColor = accent;

void Modules::onFrameRendered() noexcept {
    const auto& cfg = getConfig();
    const uint64_t totalFrames = getTelemetry().getSnapshot().totalFrames;

    // 1. Update simplified center crosshair (Module 9)
    const bool wantCrosshair =
        cfg.modHudAcceleratorEnabled.load(std::memory_order_relaxed) &&
        cfg.simplifyCrosshair.load(std::memory_order_relaxed);

    if (wantCrosshair) {
        if (!gCrosshairWasVisible || (totalFrames & 0x1Fu) == 0u) {
            const float r = static_cast<float>(
                std::clamp(cfg.crosshairSize.load(std::memory_order_relaxed), 4, 24));
            const int color = static_cast<int>(cfg.accentColorArgb.load(std::memory_order_relaxed));

            // LeviLaunchroid HudOverlay.java screen-center coordinate convention: -20000.0f + offset
            std::vector<pl::modmenu::DrawCommand> crossCmds(2);
            crossCmds[0].type  = pl::modmenu::DrawCommandType::Line;
            crossCmds[0].x     = -20000.0f - r;
            crossCmds[0].y     = -20000.0f;
            crossCmds[0].w     = r * 2.0f;
            crossCmds[0].h     = 0.0f;
            crossCmds[0].size  = 2.0f;
            crossCmds[0].color = color;

            crossCmds[1].type  = pl::modmenu::DrawCommandType::Line;
            crossCmds[1].x     = -20000.0f;
            crossCmds[1].y     = -20000.0f - r;
            crossCmds[1].w     = 0.0f;
            crossCmds[1].h     = r * 2.0f;
            crossCmds[1].size  = 2.0f;
            crossCmds[1].color = color;

            pl::modmenu::submitDrawCommands(ModuleIds::kHudUiAccelerator, crossCmds);
            gCrosshairWasVisible = true;
        }
    } else if (gCrosshairWasVisible) {
        pl::modmenu::submitDrawCommands(ModuleIds::kHudUiAccelerator, {});
        gCrosshairWasVisible = false;
    }

    // 2. Update 90-FPS Telemetry HUD (Module 12) every 12 frames (~133ms at 90 FPS)
    const bool wantHud =
        cfg.modAdaptiveGovernorEnabled.load(std::memory_order_relaxed) &&
        cfg.showFpsHud.load(std::memory_order_relaxed);

    if (!wantHud) {
        if (gHudWasVisible) {
            pl::modmenu::submitDrawCommands(ModuleIds::kAdaptiveGovernorHud, {});
            pl::modmenu::submitHudEditorElements(ModuleIds::kAdaptiveGovernorHud, {});
            gHudWasVisible = false;
        }
        return;
    }

    if (gHudWasVisible && (totalFrames - gLastHudUpdateFrame) < 12) {
        return;
    }
    gLastHudUpdateFrame = totalFrames;

    const auto cmds = buildHudDrawCommands();
    pl::modmenu::submitDrawCommands(ModuleIds::kAdaptiveGovernorHud, cmds);

    // Register movable HUD Editor bounding box so users can drag & snap the HUD
    pl::modmenu::HudEditorElement elem{};
    elem.elementId     = "hyper90.hud.telemetry_card";
    elem.displayName   = "Hyper90 Telemetry HUD";
    elem.positionKeyX  = "hudPosX";
    elem.positionKeyY  = "hudPosY";
    elem.snapGroup     = "hyper90";
    elem.x             = std::clamp(cfg.hudPosX.load(std::memory_order_relaxed), 0.0f, 3840.0f);
    elem.y             = std::clamp(cfg.hudPosY.load(std::memory_order_relaxed), 0.0f, 2160.0f);
    elem.width         = 340.0f;
    elem.height        = 142.0f;
    elem.gridSize      = 8.0f;
    elem.snapThreshold = 10.0f;
    elem.gridGap       = 6.0f;
    elem.snapFlags     = pl::modmenu::HudSnapGrid |
                         pl::modmenu::HudSnapElements |
                         pl::modmenu::HudSnapScreenCenter;

    std::vector<pl::modmenu::HudEditorElement> elements{elem};
    pl::modmenu::submitHudEditorElements(ModuleIds::kAdaptiveGovernorHud, elements);
    gHudWasVisible = true;
}

bool registerModule(const ModuleInfo& info) {
    assert(!info.moduleId.empty() && info.moduleId.size() <= 160);
    assert(!info.displayName.empty() && info.displayName.size() <= 160);
    assert(info.description.size() <= 1024);
    assert(!info.modId.empty() && info.modId.size() <= 160);
    for (const auto& cfg : info.configs) {
        assert(!cfg.key.empty() && cfg.key.size() <= 160);
        assert(!cfg.displayName.empty() && cfg.displayName.size() <= 160);
        assert(cfg.defaultValue.size() <= 2048);
    }
    gMockModules.push_back(info);
    return true;
}

            gMockModules.erase(it);

    }

    assert(!info.buttonId.empty() && info.buttonId.size() <= 160);
    assert(!info.displayName.empty() && info.displayName.size() <= 160);
    assert(info.label.size() <= 32);
    gMockButtons.push_back(info);
    return true;
}

void unregisterButton(std::string_view buttonId) {
    for (auto it = gMockButtons.begin(); it != gMockButtons.end(); ++it) {
        if (it->buttonId == buttonId) {
            gMockButtons.erase(it);
            return;
        }
    }
}

void submitDrawCommands(std::string_view moduleId, std::span<const DrawCommand> commands) {
    assert(!moduleId.empty());
    for (const auto& cmd : commands) {
        assert(std::isfinite(cmd.x) && std::isfinite(cmd.y));
        assert(std::isfinite(cmd.w) && std::isfinite(cmd.h));
        assert(std::isfinite(cmd.x3) && std::isfinite(cmd.y3));
        assert(std::isfinite(cmd.size));
        assert(cmd.text.size() <= 512);
    }
    gMockDrawCmds.assign(commands.begin(), commands.end());
}

void submitHudEditorElements(std::string_view moduleId, std::span<const HudEditorElement> elements) {
    assert(!moduleId.empty());
    for (const auto& el : elements) {
        assert(!el.elementId.empty() && el.elementId.size() <= 160);
        assert(std::isfinite(el.x) && std::isfinite(el.y));
        assert(std::isfinite(el.width) && el.width >= 1.0f);
        assert(std::isfinite(el.height) && el.height >= 1.0f);
    }
    gMockHudElements.assign(elements.begin(), elements.end());
}

// ============================================================================
// Verification Suite for All 13 Optimization Modules
// ============================================================================
int main(int argc, char** argv) {
    using namespace hyper90;

    // Optional CLI flag to generate default config/config.json & config/config.schema.json
    if (argc >= 3 && std::string_view(argv[1]) == "--generate-config") {
        const std::filesystem::path outDir(argv[2]);
        assert(saveConfigToDisk(outDir));
        assert(writeConfigSchemaToDisk(outDir));
        std::cout << "[Hyper90] Generated config.json and config.schema.json in " << outDir << "\n";
        return 0;
    }

    const std::filesystem::path tmpRoot = std::filesystem::temp_directory_path() / "hyper90_test_env";
    std::filesystem::remove_all(tmpRoot);

    // Create mock LeviLaunchroid directory layout:
    // <minecraftRoot>/<profileId>/mods/hyper90_fps_booster
    // <minecraftRoot>/<profileId>/external/games/com.mojang/minecraftpe/options.txt
    const auto profileDir = tmpRoot / "default_profile";
    const auto modDir     = profileDir / "mods" / "hyper90_fps_booster";
    const auto mcpeDir    = profileDir / "external" / "games" / "com.mojang" / "minecraftpe";
    std::filesystem::create_directories(modDir / "config");
    std::filesystem::create_directories(mcpeDir);

    const auto optionsTxtPath = mcpeDir / "options.txt";
    {
        std::ofstream ofs(optionsTxtPath);
        ofs << "gfx_vsync:1\n"
            << "gfx_max_framerate:60\n"
            << "gfx_msaa:4\n"
            << "gfx_texel_aa:1\n"
            << "gfx_fancygraphics:1\n"
            << "gfx_transparentleaves:1\n"
            << "gfx_fancyskies:1\n"
            << "gfx_renderclouds:1\n"
            << "gfx_smoothlighting:1\n"
            << "gfx_viewdistance:256\n"
            << "mp_username:Steve\n";
    }

    // ------------------------------------------------------------------------
    // Test 1: Register All 13 Modules & 2 Quick-Action Buttons
    // ------------------------------------------------------------------------
    assert(Modules::registerAll("hyper90_fps_booster", modDir));
    assert(gMockModules.size() == 13);
    assert(gMockButtons.size() == 2);
    std::cout << "[PASS] Registered all 13 ModMenu modules and 2 quick-action buttons.\n";

    // ------------------------------------------------------------------------
    // Test 2: Module 1 (90 FPS Unlocker & Frame Pacer) + Module 12 (Telemetry & Governor)
    // ------------------------------------------------------------------------
    auto& telemetry = getTelemetry();
    telemetry.resetCounters();

    // Simulate 30 frames at 90 FPS (11,111,111 ns per frame)
    for (int i = 0; i < 30; ++i) {
        telemetry.recordFrameDurationNs(11'111'111LL);
    }
    auto snap90 = telemetry.getSnapshot();
    assert(std::fabs(snap90.avgFps - 90.0f) < 0.5f);
    assert(std::fabs(snap90.onePercentLowFps - 90.0f) < 0.5f);
    assert(snap90.jitterMs < 0.05f);
    assert(snap90.tier == GovernorTier::Locked90Plus);

    // Simulate a heavy combat/chunk-load dip to 65 FPS (15,384,615 ns)
    for (int i = 0; i < 40; ++i) {
        telemetry.recordFrameDurationNs(15'384'615LL);
    }
    auto snapDip = telemetry.getSnapshot();
    assert(snapDip.avgFps < 75.0f);
    assert(snapDip.tier == GovernorTier::Ultra90Lock);
    std::cout << "[PASS] Module 1 & 12: 90-FPS Telemetry & 4-Tier Adaptive Governor verified.\n";

    // ------------------------------------------------------------------------
    // Test 3: Module 2 (Bedrock Engine & RenderDragon options.txt Tuner)
    // ------------------------------------------------------------------------
    const int tunedFiles = EngineOptionsTuner::discoverAndTuneAll(modDir);
    assert(tunedFiles == 1);
    assert(std::filesystem::exists(optionsTxtPath.string() + ".hyper90.bak"));

    {
        std::ifstream ifs(optionsTxtPath);
        std::stringstream buf;
        buf << ifs.rdbuf();
        const std::string content = buf.str();
        assert(content.find("gfx_vsync:0\n") != std::string::npos);
        assert(content.find("gfx_max_framerate:90\n") != std::string::npos);
        assert(content.find("gfx_msaa:1\n") != std::string::npos);
        assert(content.find("gfx_texel_aa:0\n") != std::string::npos);
        assert(content.find("gfx_fancygraphics:0\n") != std::string::npos);
        assert(content.find("gfx_transparentleaves:0\n") != std::string::npos);
        assert(content.find("gfx_fancyskies:0\n") != std::string::npos);
        assert(content.find("gfx_renderclouds:0\n") != std::string::npos);
        assert(content.find("gfx_smoothlighting:0\n") != std::string::npos);
        assert(content.find("gfx_viewdistance:112\n") != std::string::npos);
        assert(content.find("mp_username:Steve\n") != std::string::npos);
    }
    std::cout << "[PASS] Module 2: Bedrock options.txt auto-discovery, tuning & backup verified.\n";

    // ------------------------------------------------------------------------
    // Test 4: Module 3 (Big.LITTLE Prime Core & Thread Booster)
    // ------------------------------------------------------------------------
    // Simulate 8-core Snapdragon (3x Silver @ 1.8GHz, 4x Gold @ 2.8GHz, 1x Prime @ 3.2GHz)
    const std::vector<int64_t> snapdragonFreqs = {
        1800000, 1800000, 1800000,
        2800000, 2800000, 2800000, 2800000,
        3200000
    };
    const auto perfCores = CpuSchedulerOptimizer::selectPerformanceCores(snapdragonFreqs, 0);
    assert(perfCores.size() == 5);
    assert(perfCores[0] == 3 && perfCores[4] == 7);
    assert(CpuSchedulerOptimizer::applyRenderThreadBoost(true) > 0);
    std::cout << "[PASS] Module 3: Big.LITTLE Prime/Gold core selection & thread affinity verified.\n";

    // ------------------------------------------------------------------------
    // Test 5: Module 4 (OpenGL ES Pipeline & State Deduplicator)
    // ------------------------------------------------------------------------
    auto& glOpt = getGlOptimizer();
    glOpt.resetFrameState();

    // Shader hints rewritten to GL_FASTEST
    assert(glOpt.rewriteHintMode(GlConsts::kGlGenerateMipmapHint, 0x1102u) == GlConsts::kGlFastest);
    assert(glOpt.rewriteHintMode(GlConsts::kGlFragmentShaderDerivativeHint, 0x1100u) == GlConsts::kGlFastest);

    // GL_DITHER and GL_MULTISAMPLE_EXT blocked
    assert(glOpt.shouldSkipCapabilityChange(GlConsts::kGlDither, true) == true);
    assert(glOpt.shouldSkipCapabilityChange(GlConsts::kGlMultisampleExt, true) == true);

    // State deduplication for glEnable(GL_BLEND), glUseProgram, glBindTexture, glDepthMask
    assert(glOpt.shouldSkipCapabilityChange(GlConsts::kGlBlend, true) == false);
    assert(glOpt.shouldSkipCapabilityChange(GlConsts::kGlBlend, true) == true); // Deduplicated!
    assert(glOpt.shouldSkipUseProgram(42u) == false);
    assert(glOpt.shouldSkipUseProgram(42u) == true); // Deduplicated!
    glOpt.onActiveTexture(GlConsts::kGlTexture0 + 1);
    assert(glOpt.shouldSkipBindTexture(GlConsts::kGlTexture2D, 99u) == false);
    assert(glOpt.shouldSkipBindTexture(GlConsts::kGlTexture2D, 99u) == true); // Deduplicated!
    assert(glOpt.shouldSkipDepthMask(1) == false);
    assert(glOpt.shouldSkipDepthMask(1) == true); // Deduplicated!
    std::cout << "[PASS] Module 4: OpenGL ES Pipeline & State Deduplicator verified.\n";

    // ------------------------------------------------------------------------
    // Test 6: Module 5 (Adreno/Mali TBDR Tile Bandwidth Saver)
    // ------------------------------------------------------------------------
    const uint32_t mergedMask = glOpt.rewriteClearMask(GlConsts::kGlDepthBufferBit);
    assert((mergedMask & GlConsts::kGlDepthBufferBit) != 0u);
    assert((mergedMask & GlConsts::kGlStencilBufferBit) != 0u);
    std::cout << "[PASS] Module 5: Adreno/Mali TBDR Tile Bandwidth Saver verified.\n";

    // ------------------------------------------------------------------------
    // Test 7: Module 6 (VRAM & Texture Mipmap Optimizer)
    // ------------------------------------------------------------------------
    int32_t minFilter = static_cast<int32_t>(GlConsts::kGlLinearMipmapLinear);
    glOpt.rewriteTexParameteri(GlConsts::kGlTexture2D, GlConsts::kGlTextureMinFilter, minFilter);
    assert(static_cast<uint32_t>(minFilter) != GlConsts::kGlLinearMipmapLinear);

    float aniso = 16.0f;
    glOpt.rewriteTexParameterf(GlConsts::kGlTexture2D, GlConsts::kGlTextureMaxAnisotropyExt, aniso);
    assert(aniso == 1.0f);
    std::cout << "[PASS] Module 6: VRAM & Texture Mipmap Optimizer verified.\n";

    // ------------------------------------------------------------------------
    // Test 8: Module 7 & Module 8 (Draw Batch Culler & Entity/Particle Guard)
    // ------------------------------------------------------------------------
    // Degenerate 0-index draw call is always culled
    assert(glOpt.shouldCullDrawCall(GlConsts::kGlTriangles, 0) == true);

    // Submit 90 terrain draw calls (300 indices each -> never culled)
    for (int i = 0; i < 90; ++i) {
        assert(glOpt.shouldCullDrawCall(GlConsts::kGlTriangles, 300) == false);
    }

    // Submit 200 micro-batch particle draws (6 indices each -> adaptive & budget culling kicks in)
    int culledParticles = 0;
    for (int i = 0; i < 200; ++i) {
        if (glOpt.shouldCullDrawCall(GlConsts::kGlTriangles, 6)) {
            ++culledParticles;
        }
    }
    assert(culledParticles > 50);
    std::cout << "[PASS] Module 7 & 8: Degenerate, Micro-Batch & Particle Budget Culler verified.\n";

    // ------------------------------------------------------------------------
    // Test 9: Module 9 (Zero-Alloc HUD & UI Batch Accelerator)
    // ------------------------------------------------------------------------
    auto& hudCache = getHudCache();
    hudCache.invalidateAll();
    HotbarSlotState slot0{276, 0, 1, 0x1234u};
    assert(hudCache.checkAndUpdateHotbarSlot(0, slot0) == false); // First time: miss
    assert(hudCache.checkAndUpdateHotbarSlot(0, slot0) == true);  // Second time: hit!

    PlayerVitalsState vitals{20, 20, 20, 15, 30, 500};
    assert(hudCache.checkAndUpdateVitals(vitals) == false);
    assert(hudCache.checkAndUpdateVitals(vitals) == true);

    assert(hudCache.checkAndUpdateText(1, "Position: 120, 64, -300") == false);
    assert(hudCache.checkAndUpdateText(1, "Position: 120, 64, -300") == true);
    std::cout << "[PASS] Module 9: Zero-Alloc HUD Hotbar, Vitals & FNV-1a Text Cache verified.\n";

    // ------------------------------------------------------------------------
    // Test 10: Module 10 (Bionic Heap & Page-Fault Stutter Shield)
    // ------------------------------------------------------------------------
    (void)MemoryArenaOptimizer::applyBionicHeapTuning();
    (void)MemoryArenaOptimizer::lockHotMemoryPages();
    MemoryArenaOptimizer::releaseLockedPages();
    std::cout << "[PASS] Module 10: Bionic Heap & Page-Fault Stutter Shield verified.\n";

    // ------------------------------------------------------------------------
    // Test 11: Module 11 (90Hz Sub-Frame Camera & Touch Stabilizer)
    // ------------------------------------------------------------------------
    auto& cam = getCameraStabilizer();
    cam.reset();
    float dx = 4.0f, dy = 2.0f;
    cam.filterTurnDelta(dx, dy, 0.0111f); // Initial sample
    float dx2 = 6.0f, dy2 = 3.0f;
    cam.filterTurnDelta(dx2, dy2, 0.0111f);
    assert(dx2 > 4.0f && dx2 < 6.0f); // Smoothly interpolated at 90Hz!

    // Raw Flick Bypass (> 30 deg flick passes through unattenuated)
    float flickX = 45.0f, flickY = 0.0f;
    cam.filterTurnDelta(flickX, flickY, 0.0111f);
    assert(std::fabs(flickX - 45.0f) < 1e-4f);
    std::cout << "[PASS] Module 11: 90Hz Sub-Frame Camera Filter & Raw Flick Bypass verified.\n";

    // ------------------------------------------------------------------------
    // Test 12: Live HUD DrawCommands, HUD Editor Elements & Config Persistence
    // ------------------------------------------------------------------------
    Modules::handleConfigChange(ModuleIds::kHudUiAccelerator, "simplify_crosshair", "true");
    Modules::onFrameRendered();
    assert(!gMockDrawCmds.empty());
    assert(!gMockHudElements.empty());

    // Toggle & config change round-trip persistence test
    Modules::handleConfigChange(ModuleIds::kFramePacer, "target_fps", "90");
    assert(loadConfigFromDisk(modDir / "config"));
    assert(getConfig().targetFps.load() == 90);

    // ------------------------------------------------------------------------
    // Test 13: Module 13 (Frame Blend) - planner, telemetry accounting, config
    // ------------------------------------------------------------------------
    {
        using namespace hyper90::FrameBlendConsts;

        // Off by default (experimental, forces VSync while active)
        assert(!getConfig().modFrameBlendEnabled.load());
        assert(!FrameBlendGl::wantsVsync());

        // Multiplier & weight helpers
        assert(std::fabs(FrameBlendPlanner::multiplierForMode(kModeX15) - 1.5f) < 1e-6f);
        assert(std::fabs(FrameBlendPlanner::multiplierForMode(kModeX20) - 2.0f) < 1e-6f);
        assert(std::fabs(FrameBlendPlanner::weightForStrengthPct(50) - 0.5f) < 1e-6f);
        assert(std::fabs(FrameBlendPlanner::weightForStrengthPct(0) - 0.1f) < 1e-6f);
        assert(std::fabs(FrameBlendPlanner::weightForStrengthPct(500) - 0.9f) < 1e-6f);

        // 2.0x: every real frame generates one (after the first, which has no history)
        FrameBlendPlanner planner;
        assert(!planner.planFrame(kModeX20, 22'000'000LL, false).generate);
        for (int i = 0; i < 10; ++i) {
            const auto plan = planner.planFrame(kModeX20, 22'000'000LL, true);
            assert(plan.generate && plan.presentsThisFrame == 2);
        }

        // 1.5x: 3 presents every 2 real frames
        planner.reset();
        int generated = 0;
        for (int i = 0; i < 100; ++i) {
            generated += planner.planFrame(kModeX15, 16'000'000LL, true).generate ? 1 : 0;
        }
        assert(generated == 50);

        // Too-slow real frames (< 20 FPS) are never blended; invalid intervals neither
        assert(!planner.planFrame(kModeX20, 80'000'000LL, true).generate);
        assert(!planner.planFrame(kModeX20, 0, true).generate);

        // Telemetry: 45 real FPS with one generated frame each must read as 90 displayed FPS
        telemetry.resetCounters();
        for (int i = 0; i < 40; ++i) {
            telemetry.recordGeneratedFrameDurationNs(11'111'111LL);
            telemetry.recordFrameDurationNs(11'111'111LL);
        }
        const auto snapBlend = telemetry.getSnapshot();
        assert(std::fabs(snapBlend.avgFps - 90.0f) < 0.5f);
        assert(snapBlend.generatedFrames == 40);
        assert(snapBlend.totalFrames == 40); // generated frames do not count as rendered frames

        // Config persistence round-trip (mode / strength / enabled)
        Modules::handleModuleToggle(ModuleIds::kFrameBlend, true);
        Modules::handleConfigChange(ModuleIds::kFrameBlend, "mode", "0");
        Modules::handleConfigChange(ModuleIds::kFrameBlend, "strength_pct", "65");
        assert(FrameBlendGl::wantsVsync());
        getConfig().modFrameBlendEnabled.store(false);
        getConfig().frameBlendMode.store(1);
        getConfig().frameBlendStrengthPct.store(50);
        assert(loadConfigFromDisk(modDir / "config"));
        assert(getConfig().modFrameBlendEnabled.load());
        assert(getConfig().frameBlendMode.load() == 0);
        assert(getConfig().frameBlendStrengthPct.load() == 65);

        // Out-of-range values are clamped
        Modules::handleConfigChange(ModuleIds::kFrameBlend, "strength_pct", "999");
        assert(getConfig().frameBlendStrengthPct.load() == 90);

        // Leave the module off so the remaining tests are unaffected
        Modules::handleModuleToggle(ModuleIds::kFrameBlend, false);
        assert(!FrameBlendGl::wantsVsync());
        assert(FrameBlendGl::onBeforeSwap(nullptr, nullptr, FrameBlendPresentOps{}) == 0);
        std::cout << "[PASS] Module 13: Frame Blend planner, displayed-FPS accounting & config verified.\n";
    }

    Modules::unregisterAll();
    assert(gMockModules.empty());
    assert(gMockButtons.empty());

    std::filesystem::remove_all(tmpRoot);
    std::cout << "[ALL 13 MODULES PASSED] Hyper90 FPS Booster verification complete!\n";
    return 0;
}
